% =============================================================================
%  ATU PRESENTATION THEME: single source of truth for all visual styling
%
%  Everything below derives from the geometry block in section 2.
%  Change a length there and the sidebar, margins, title box, and every
%  decoration on the title / thank-you slides all follow automatically.
% =============================================================================


% =============================================================================
%  1. COLOUR PALETTE
% =============================================================================
% --- Pastels: BACKGROUNDS ONLY ----------------------------------------------
% These are light by construction, so white text on them is unreadable
% (2.1-2.7:1). Never set white or another pastel as foreground on these.
\definecolor{BabyBlue}{RGB}{137,207,240}
\definecolor{BabyPink}{RGB}{244,194,194}
\definecolor{BabyMint}{RGB}{170,225,200}
\definecolor{BabyLavender}{RGB}{200,180,220}
\definecolor{BabyPeach}{RGB}{255,218,185}
\definecolor{SoftTeal}{RGB}{90,170,175}

% --- Deep tones: FOREGROUND / TEXT ------------------------------------------
% Saturated partners of the pastels above. Each pairs with its own pastel at
% >= 4.9:1 on the tinted fills, and >= 5.8:1 on white. (WCAG AA is 4.5:1.)
\definecolor{DeepTeal}{RGB}{0,91,94}
\definecolor{DeepGreen}{RGB}{21,115,71}
\definecolor{DeepRed}{RGB}{178,45,58}
\definecolor{DeepPurple}{RGB}{94,63,158}
\definecolor{DarkText}{RGB}{40,40,50}
\definecolor{MidGrey}{RGB}{95,95,108}   % darkened: was 120,120,130 = 4.37:1


% =============================================================================
%  2. GEOMETRY : the only numbers you should ever need to edit
% =============================================================================
\newlength{\atuBar}        \setlength{\atuBar}{0.85cm}   % teal sidebar
\newlength{\atuStripe}     \setlength{\atuStripe}{0.06cm}% accent stripe
\newlength{\atuGutter}     \setlength{\atuGutter}{0.55cm}% bar -> content gap
\newlength{\atuMarginR}    \setlength{\atuMarginR}{0.6cm}

% --- Derived (do not edit) ---------------------------------------------------
\newlength{\atuBarTotal}   % sidebar incl. stripe: the "no-go" zone
\setlength{\atuBarTotal}{\dimexpr\atuBar+\atuStripe\relax}

\newlength{\atuTextLeft}   % left text margin
\setlength{\atuTextLeft}{\dimexpr\atuBarTotal+\atuGutter\relax}

\newlength{\atuContentWidth}  % usable width for ALL content incl. title box
\setlength{\atuContentWidth}{%
  \dimexpr\paperwidth-\atuTextLeft-\atuMarginR\relax}

\setbeamersize{text margin left=\atuTextLeft, text margin right=\atuMarginR}
\setlength{\parskip}{2pt}


% =============================================================================
%  3. BEAMER BASE THEME
% =============================================================================
\usetheme{default}
\usecolortheme{default}
\usefonttheme{professionalfonts}

\setbeamercolor{background canvas}{bg=white}
\setbeamercolor{normal text}{fg=DarkText}
\setbeamercolor{frametitle}{fg=DeepTeal,bg=white}
\setbeamercolor{title}{fg=DeepTeal}
\setbeamercolor{subtitle}{fg=SoftTeal}
\setbeamercolor{author}{fg=DarkText}
\setbeamercolor{institute}{fg=MidGrey}
\setbeamercolor{date}{fg=MidGrey}
\setbeamercolor{structure}{fg=DeepTeal}
\setbeamercolor{itemize item}{fg=DeepTeal}
\setbeamercolor{itemize subitem}{fg=MidGrey}
\setbeamercolor{enumerate item}{fg=DeepTeal}

% Block titles: pastel fill + deep same-hue text (was white-on-pastel, 2.1-2.7:1)
\setbeamercolor{block title}{fg=DeepTeal,bg=BabyBlue!35}
\setbeamercolor{block body}{fg=DarkText,bg=BabyBlue!12}
\setbeamercolor{block title alerted}{fg=DeepRed,bg=BabyPink!45}
\setbeamercolor{block body alerted}{fg=DarkText,bg=BabyPink!15}
\setbeamercolor{block title example}{fg=DeepGreen,bg=BabyMint!45}
\setbeamercolor{block body example}{fg=DarkText,bg=BabyMint!15}

\setbeamerfont{frametitle}{size=\large,series=\bfseries}
\setbeamerfont{block title}{size=\normalsize,series=\bfseries}

% Flat design: sharp corners, solid fills, no shadows or bevels anywhere.
\setbeamertemplate{blocks}[rectangular]
% NB: these templates hardcode the colour, so they OVERRIDE the
% \setbeamercolor lines above. Keep both in sync or the bullets silently
% revert to a low-contrast pastel.
\setbeamertemplate{itemize item}{%
  \small\raise1pt\hbox{\textcolor{DeepTeal}{\textbullet}}}
\setbeamertemplate{itemize subitem}{%
  \scriptsize\raise1pt\hbox{\textcolor{MidGrey}{\textendash}}}


% =============================================================================
%  4. SIDEBAR : drawn on every content slide
% =============================================================================
% Reusable sidebar drawing, so content slides and special slides stay in sync.
\newcommand{\atudrawsidebar}{%
  \fill[DeepTeal]
    (current page.north west) rectangle
    ([xshift=\atuBar]current page.south west);
  \fill[BabyBlue]
    ([xshift=\atuBar]current page.north west) rectangle
    ([xshift=\atuBarTotal]current page.south west);%
}

\setbeamertemplate{background}{%
  \begin{tikzpicture}[remember picture,overlay]
    \atudrawsidebar
    \node[anchor=south, font=\scriptsize\bfseries, text=white]
      at ([xshift=0.5\atuBar, yshift=7pt]current page.south west)
      {\insertframenumber\,/\,\inserttotalframenumber};
  \end{tikzpicture}%
}


% =============================================================================
%  5. FRAME TITLE : flat design
%     No outline, no rounded corners, no rules-as-decoration. A solid tint
%     band spanning the full content width, anchored by a solid accent tab.
%     Width is exactly \atuContentWidth, so the right edge always lands on
%     the right margin and can never be clipped by the page edge.
% =============================================================================
\newlength{\atuTitleTab}   \setlength{\atuTitleTab}{5pt}   % accent tab
\newlength{\atuTitleInset} \setlength{\atuTitleInset}{15pt}% text inset

\setbeamertemplate{frametitle}{%
  \vskip6pt%
  \begin{tcolorbox}[
    enhanced, sharp corners, frame hidden, boxrule=0pt,
    colback=DeepTeal!7,
    left=\atuTitleInset, right=10pt, top=6pt, bottom=6pt,
    width=\atuContentWidth,
    borderline west={\atuTitleTab}{0pt}{DeepTeal}]%
    {\usebeamerfont{frametitle}\color{DeepTeal}\insertframetitle}%
  \end{tcolorbox}%
  \vskip8pt%
}


% =============================================================================
%  6. FOOTER
% =============================================================================
\setbeamertemplate{navigation symbols}{}
\setbeamertemplate{headline}{}
% The \rule gives the footline a real height so beamer reserves this space and
% content can never run into the logo or the page number.
\newlength{\atuFooterHeight}\setlength{\atuFooterHeight}{0.55cm}
\setbeamertemplate{footline}{%
  \begin{tikzpicture}[remember picture,overlay]
    \node[anchor=south east]
      at ([xshift=-\atuMarginR, yshift=5pt]current page.south east)
      {\includegraphics[height=0.32cm]{images/atu-logo-teal.png}};
  \end{tikzpicture}%
  \rule{0pt}{\atuFooterHeight}%
}

\newcommand{\tablestyle}{\renewcommand{\arraystretch}{1.2}\small}

% Status tags. \mbox keeps each on one line: must never hyphenate on a wrap.
% Deep tones, not pastels: these sit on white and need >= 4.5:1.

\newcommand{\rejected}{\mbox{{\small\color{DeepRed}(REJECTED)}}}
\newcommand{\novelfinding}{\mbox{{\small\color{DeepPurple}(Novel Finding)}}}

% Same tags for use inside tables.
\newcommand{\tagconfirmed}{\textcolor{DeepGreen}{\textbf{CONFIRMED}}}
\newcommand{\tagrejected}{\textcolor{DeepRed}{\textbf{REJECTED}}}

% --- Columns that respect the text margins ----------------------------------
% Plain \begin{columns} centres its block on the PAGE, ignoring the asymmetric
% margins, which pushes content left into the sidebar. Always use this instead.
\newenvironment{atucols}[1][c]
  {\begin{columns}[#1,onlytextwidth]}
  {\end{columns}}

% --- Full-bleed background for title / thank-you slides ---------------------
% Bookend slides: NO sidebar, full-bleed tint.
% #1 = extra TikZ decorations, drawn over the base tint.
% NB: do NOT call \setbeamersize here. Between frames it runs in horizontal
% mode and beamer echoes the unmatched key as text, emitting a blank page.
% Bookend content is centred with \atupagetitle instead, which is immune to
% whatever the margins happen to be.
\newcommand{\atuspecialbg}[1]{%
  \setbeamertemplate{headline}{}%
  \setbeamertemplate{footline}{}%
  \setbeamertemplate{background}{%
    \begin{tikzpicture}[remember picture,overlay]
      \fill[BabyBlue!8]
        (current page.north west) rectangle (current page.south east);
      #1
      \atudrawsidebar
    \end{tikzpicture}%
  }%
}

% --- Centred block for bookend slides ---------------------------------------
% Anchored to the page centre, so it is optically centred regardless of the
% text margins. \atupagetitle{<content>}
\newcommand{\atupagetitle}[1]{%
  \node[anchor=center, align=center,
        text width=\dimexpr\paperwidth-\atuBarTotal-2cm\relax]
    at ([xshift=0.5\atuBarTotal, yshift=0.2cm]current page.center) {#1};%
}

% --- Accent rule at a given y-offset from an anchor --------------------------
% \aturule[<left inset>]{<anchor>}{<yshift>}{<colour>}{<width>}
% Default inset clears the sidebar; pass \atuMarginR on sidebar-less slides.
\newcommand{\aturule}[5][\atuBarTotal]{%
  \draw[#4, line width=#5]
    ([xshift=#1, yshift=#3]current page.#2 west) --
    ([xshift=-\atuMarginR, yshift=#3]current page.#2 east);%
}

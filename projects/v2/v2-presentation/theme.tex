% =============================================================================
%  1. COLOUR PALETTE
% =============================================================================

% =============================================================================
%  2. GEOMETRY  — the only numbers you should ever need to edit
% =============================================================================
\newlength{\atuBar}        \setlength{\atuBar}{0.85cm}   % teal sidebar
\newlength{\atuStripe}     \setlength{\atuStripe}{0.06cm}% accent stripe
\newlength{\atuGutter}     \setlength{\atuGutter}{0.55cm}% bar -> content gap
\newlength{\atuMarginR}    \setlength{\atuMarginR}{0.6cm}

% --- Derived (do not edit) ---------------------------------------------------
\newlength{\atuBarTotal}   % sidebar incl. stripe — the "no-go" zone
\setlength{\atuBarTotal}{\dimexpr\atuBar+\atuStripe\relax}

\newlength{\atuTextLeft}   % left text margin
\setlength{\atuTextLeft}{\dimexpr\atuBarTotal+\atuGutter\relax}

\newlength{\atuContentWidth}  % usable width for ALL content incl. title box
\setlength{\atuContentWidth}{%
  \dimexpr\paperwidth-\atuTextLeft-\atuMarginR\relax}

\setbeamersize{text margin left=\atuTextLeft, text margin right=\atuMarginR}
\setlength{\parskip}{2pt}

% =============================================================================
%  4. SIDEBAR  — drawn on every content slide
% =============================================================================
% Reusable sidebar drawing, so content slides and special slides stay in sync.
\newcommand{\atudrawsidebar}{%
  \fill[DeepTeal]
    (current page.north west) rectangle
    ([xshift=\atuBar]current page.south west);
  \fill[BabyBlue]
    ([xshift=\atuBar]current page.north west) rectangle
    ([xshift=\atuBarTotal]current page.south west);%
}

\setbeamertemplate{background}{%
  \begin{tikzpicture}[remember picture,overlay]
    \atudrawsidebar
    \node[anchor=south, font=\scriptsize\bfseries, text=white]
      at ([xshift=0.5\atuBar, yshift=7pt]current page.south west)
      {\insertframenumber\,/\,\inserttotalframenumber};
  \end{tikzpicture}%
}


% =============================================================================
%  5. FRAME TITLE  — boxed, with full content-width underline
%     Width is exactly \atuContentWidth, so the right edge always lands on
%     the right margin and can never be clipped by the page edge.
% =============================================================================
\setbeamertemplate{frametitle}{%
  \vskip5pt%
  \begin{tcolorbox}[
    enhanced, colback=white, colframe=DeepTeal,
    boxrule=0.7pt, arc=2pt,
    left=8pt, right=8pt, top=3.5pt, bottom=3.5pt,
    width=\atuContentWidth,
    borderline south={2pt}{-2.4pt}{BabyBlue}]%
    {\usebeamerfont{frametitle}\color{DeepTeal}\insertframetitle}%
  \end{tcolorbox}%
  \vskip6pt%
}

% --- Columns that respect the text margins ----------------------------------
% Plain \begin{columns} centres its block on the PAGE, ignoring the asymmetric
% margins, which pushes content left into the sidebar. Always use this instead.
\newenvironment{atucols}[1][c]
  {\begin{columns}[#1,onlytextwidth]}
  {\end{columns}}

% --- Full-bleed background for title / thank-you slides ---------------------
% #1 = extra TikZ decorations. Decorations must start at \atuBarTotal so they
% never paint over the sidebar.
\newcommand{\atuspecialbg}[1]{%
  \setbeamertemplate{headline}{}%
  \setbeamertemplate{footline}{}%
  \setbeamertemplate{background}{%
    \begin{tikzpicture}[remember picture,overlay]
      \fill[BabyBlue!8]
        (current page.north west) rectangle (current page.south east);
      #1
      \atudrawsidebar
    \end{tikzpicture}%
  }%
}

% --- Accent rule spanning content width, at a given y-offset from an anchor --
% \aturule{<anchor>}{<yshift>}{<colour>}{<width>}
\newcommand{\aturule}[4]{%
  \draw[#3, line width=#4]
    ([xshift=\atuBarTotal, yshift=#2]current page.#1 west) --
    ([yshift=#2]current page.#1 east);%
}

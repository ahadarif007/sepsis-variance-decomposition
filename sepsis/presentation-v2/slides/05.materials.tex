\begin{frame}{IV. Materials: Study Design}
\begin{itemize}
  \item \textbf{Development dataset:} MIMIC-IV
        v3.1~\cite{johnson2023mimiciv} (65,078 adult ICU stays)
  \item \textbf{External validation:} eICU-CRD v2.0~\cite{pollard2018eicu}
        (200+ US ICUs, 7.9 million person-hours)
\end{itemize}

\vspace{4pt}
\textbf{Three ways to define sepsis~\cite{singer2016sepsis3}:}
\vspace{1pt}
\begin{center}\renewcommand{\arraystretch}{1.1}\scriptsize
\begin{tabular}{clp{5.6cm}p{3.6cm}}
\toprule
\textbf{Variant} & \textbf{Meaning} & \textbf{When infection is confirmed}
  & \textbf{SOFA baseline} \\
\midrule
A & \textbf{Strict}
  & Antibiotics and blood culture must happen within \textbf{24 hours} of
    each other.
  & SOFA score at ICU admission \\[3pt]
B & \textbf{Standard}
  & Antibiotics can come up to \textbf{72 hours before} the blood culture,
    or \textbf{24 hours after} it.
  & SOFA score at ICU admission \\[3pt]
C & \textbf{Broad}
  & Antibiotics and blood culture can happen within \textbf{72 hours} of
    each other.
  & Lowest SOFA score in the previous 24 hours \\
\bottomrule
\end{tabular}
\end{center}

\vspace{4pt}
\begin{alertblock}{In all three definitions, a patient has sepsis if:}
  \small There is \textbf{suspected infection}, and the \textbf{SOFA (Sequential
  Organ Failure Assessment) score increases by at least 2 points}.
\end{alertblock}
\end{frame}

\begin{column}{0.51\textwidth}
\begin{itemize}\itemsep3pt\small
  \item \textbf{Development:} MIMIC-IV v3.1~\cite{johnson2023mimiciv}, a
        single centre (BIDMC, Boston).
  \item \textbf{Inclusion:} age $\geq 18$, first ICU stay per patient, length
        of stay $\geq 4$\,h $\Rightarrow$ \textbf{\pcNStays{} stays}.
  \item \textbf{External:} eICU-CRD v2.0~\cite{pollard2018eicu}, over 200 US
        ICUs, \pcExtNCohortStays{} unit stays; models applied
        \textbf{frozen, without retraining}.
  \item \textbf{Person-hour panel:} 72\,h horizon; discharge and death are
        competing events, not censoring.
  \item \textbf{22 covariates} (6 vitals, 5 labs, 2 vasopressor flags, 3
        six-hour slopes, 6 test-ordered indicators), \pcNCandidatePredictors{}
        terms per outcome.
\end{itemize}
\end{column}

\begin{atucols}[c]
\begin{column}{0.62\textwidth}
\begin{itemize}\itemsep2pt\footnotesize
  \item All three: \textbf{suspected infection} (antibiotic paired with a
        culture inside the windows) \textbf{and} a SOFA rise of $\geq 2$
        points~\cite{singer2016sepsis3,cohen2024subtle}.
  \item \textbf{Variant B is the primary analysis}; A and C are
        pre-specified alternatives.
  \item Onset is assigned at the suspicion time $t_{\text{susp}}$. The SOFA
        criterion decides which stays are septic, not when onset occurs.
\end{itemize}

\begin{frame}{VI. Results: H3, the Empty Pre-Treatment Window}
\vspace*{0pt}%
{\footnotesize Status: \notevaluable, because its contrast cannot be formed.\par}
\begin{block}{Proposition (proved in the thesis)}
  \footnotesize Under variants A, B, C (and E) the onset is $t_{\text{susp}}$,
  the \emph{later} element of an antibiotic--culture pair, so
  $t_{\text{onset}} \geq t_{\text{anchor}}$ for every labelled stay. The
  pre-treatment window therefore contains no onset \textbf{by construction}:
  a property of the implementation, not a measurement.
\end{block}

\begin{frame}{V. Methods: Models and Validation Architecture}
\begin{block}{Primary model: dynamic discrete-time competing-risks hazard}
  \footnotesize Multinomial pooled logistic regression over person-hours
  (sepsis / discharge / death), with a natural-spline baseline in elapsed
  hours~\cite{dagostino1990pooled,heyard2019dynamic}. A small ridge penalty
  addresses complete separation.
\end{block}

\vspace{2pt}
{\footnotesize\textbf{Comparators:} GBT (XGBoost), Cox regression, NEWS2,
qSOFA and SIRS~\cite{evans2021ssc}. GBT uses fixed hyperparameters and early
stopping on a 15\% stay-level training holdout.\par}

\begin{frame}{V. Methods: Metrics and Statistical Inference}
\textbf{Primary metric: adapted PhysioNet-style utility
score}~\cite{reyna2020utility}
\[
  \widetilde{U} = \frac{U - U_{\text{inact}}}{U_{\text{opt}} - U_{\text{inact}}}
  \qquad
  1\text{: perfect forecaster},\quad
  0\text{: no-alert strategy},\quad
  {<}\,0\text{: worse than no alerts}
\]
\vspace{-6pt}
\begin{itemize}\itemsep2pt\small
  \item Rewards alerts from 6\,h before onset to onset; penalises late and
        false alerts.
  \item Reported at the 0.5 threshold \textbf{and} at the best point on a
        60-point alert-rate grid.
  \item Alert burden: false alerts per true alert; published benchmark =
        1.4~\cite{moor2023review}.
  \item AUROC, AUPRC and calibration are reported alongside.
\end{itemize}

\vspace{2pt}
\begin{exampleblock}{Inference and reporting}
\begin{itemize}\itemsep1pt\footnotesize
  \item Stay-cluster bootstrap, $B = 2{,}000$ replicates ($1{,}000$ for
        eICU-CRD).
  \item Confirmatory family: Holm correction across H1--H6 (FWER 0.05).
        Exploratory analyses: Benjamini--Hochberg.
  \item Post-hoc amendments labelled explicitly, including the inference
        plan itself (Amendment 1); TRIPOD+AI and PROBAST+AI
        reporting~\cite{collins2024tripodai,moons2025probastai}.
\end{itemize}
\end{exampleblock}
\end{frame}
